@startuml
class User {
  - name: String
  - address: String
  - email: String
  - role: String
  + login()
  + updateProfile()
  + changePassword()
}

class Patient {
  - name: String
  - address: String
  - email: String
  + register()
  + updateInfo()
}

class Appointment {
  - patientId: Int
  - date: Date
  - reason: String
  + schedule()
  + reschedule()
  + cancel()
}

class Therapy {
  - date: Date
  - description: String
  - therapistId: Int
  + recordTherapy()
  + updateTherapy()
}

class MedicalRecord {
  - date: Date
  - therapyDetails: String
  - employeeId: Int
  - patientId: Int
  - appointmentDate: Date
  + addEntry()
  + updateRecord()
}

User --> Therapy
User --> MedicalRecord
Patient --> Appointment
Patient --> MedicalRecord
@enduml
